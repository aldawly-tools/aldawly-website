rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isStoreAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && (
          request.auth.token.email == "YOUR_GOOGLE_EMAIL@example.com"
          || request.auth.token.email == "YOUR_SECOND_ADMIN_EMAIL@example.com"
        );
    }

    match /products/{productId} {
      allow read: if true;
      allow write: if isStoreAdmin();
    }

    match /orders/{orderId} {
      allow create: if request.resource.data.keys().hasOnly([
          "orderNumber", "customerName", "customerPhone", "customerAddress",
          "notes", "items", "totalText", "depositAmount", "depositStatus",
          "status", "source", "createdAt", "productSubtotal", "totalAmount",
          "totalIsComplete", "whatsappStatus"
        ])
        && request.resource.data.orderNumber is string
        && request.resource.data.orderNumber.matches("^DW-[0-9]{8}$")
        && request.resource.data.status == "new"
        && request.resource.data.source == "website"
        && request.resource.data.createdAt == request.time
        && request.resource.data.customerName is string
        && request.resource.data.customerName.size() > 0
        && request.resource.data.customerName.size() <= 80
        && request.resource.data.customerPhone.matches("^[0-9]{8,15}$")
        && request.resource.data.customerAddress is string
        && request.resource.data.customerAddress.size() > 0
        && request.resource.data.customerAddress.size() <= 300
        && request.resource.data.notes is string
        && request.resource.data.notes.size() <= 300
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 50
        && request.resource.data.depositAmount == 200
        && request.resource.data.depositStatus == "pending"
        && request.resource.data.totalText is string
        && request.resource.data.totalText.size() <= 1000
        && (
          !request.resource.data.keys().hasAny([
            "productSubtotal", "totalAmount", "totalIsComplete", "whatsappStatus"
          ])
          || (
            request.resource.data.keys().hasAll([
              "productSubtotal", "totalAmount", "totalIsComplete", "whatsappStatus"
            ])
            && request.resource.data.orderNumber == orderId
            && request.resource.data.whatsappStatus == "awaiting_customer_send"
            && request.resource.data.productSubtotal is number
            && request.resource.data.productSubtotal >= 0
            && request.resource.data.totalIsComplete is bool
            && (request.resource.data.totalAmount == null
              || (request.resource.data.totalAmount is number && request.resource.data.totalAmount >= 0))
          )
        );
      allow read, update, delete: if isStoreAdmin();
    }
  }
}
