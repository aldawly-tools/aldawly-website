rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isStoreAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && (
          request.auth.token.email == "elgoharemahmoud@gmail.com"
          || request.auth.token.email == "ahmedelgohare90@gmail.com"
        );
    }

    // Visitor records contain pseudonymous per-browser activity; never expose raw rows.
    // Public counters are returned separately by the trusted HTTPS function.
    match /siteVisits/{visitId} {
      allow read: if isStoreAdmin();
      allow create: if request.resource.data.keys().hasOnly(["visitorId", "day", "createdAt"])
        && request.resource.data.visitorId is string
        && request.resource.data.visitorId.size() >= 16
        && request.resource.data.visitorId.size() <= 80
        && request.resource.data.day is string
        && request.resource.data.day.matches("^[0-9]{4}-[0-9]{2}-[0-9]{2}$")
        && request.resource.data.createdAt == request.time;
      allow update: if false;
      allow delete: if isStoreAdmin();
    }


    // Anonymous traffic-source records: one record per browser per Cairo day.
    match /trafficSources/{recordId} {
      allow create: if request.resource.data.keys().hasOnly(["visitorId", "day", "source", "createdAt"])
        && request.resource.data.visitorId is string
        && request.resource.data.visitorId.size() >= 16
        && request.resource.data.visitorId.size() <= 100
        && request.resource.data.day is string
        && request.resource.data.day.matches("^[0-9]{4}-[0-9]{2}-[0-9]{2}$")
        && request.resource.data.source is string
        && request.resource.data.source.size() <= 40
        && request.resource.data.createdAt == request.time;
      allow read: if isStoreAdmin();
      allow update: if false;
      allow delete: if isStoreAdmin();
    }

    // Anonymous product views: one record per browser, product and Cairo day.
    match /productViews/{viewId} {
      allow create: if request.resource.data.keys().hasOnly(["visitorId", "day", "productId", "createdAt"])
        && request.resource.data.visitorId is string
        && request.resource.data.visitorId.size() >= 16
        && request.resource.data.visitorId.size() <= 100
        && request.resource.data.day is string
        && request.resource.data.day.matches("^[0-9]{4}-[0-9]{2}-[0-9]{2}$")
        && request.resource.data.productId is string
        && request.resource.data.productId.size() > 0
        && request.resource.data.productId.size() <= 80
        && request.resource.data.createdAt == request.time;
      allow read: if isStoreAdmin();
      allow update: if false;
      allow delete: if isStoreAdmin();
    }

    // Public read-only aggregate counts; only the trusted Cloud Function writes them.
    match /productViewCounts/{productId} {
      allow read: if true;
      allow write: if false;
    }

    match /products/{productId} {
      allow read: if true;
      allow write: if isStoreAdmin();
    }

    match /storeMeta/{metaId} {
      allow read, write: if isStoreAdmin();
    }

    match /productImages/{imageId} {
      allow read: if true;
      allow write: if isStoreAdmin();
    }

    match /productReviews/{productId}/reviews/{reviewId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(["rating", "name", "text", "createdAt"])
        && request.resource.data.rating is int
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 40
        && request.resource.data.text is string
        && request.resource.data.text.size() <= 500
        && request.resource.data.createdAt == request.time;
      allow update, delete: if isStoreAdmin();
    }

    match /orders/{orderId} {
      allow create: if request.resource.data.keys().hasOnly([
          "orderNumber", "customerName", "customerPhone", "customerAddress",
          "notes", "items", "totalText", "status", "source", "createdAt",
          "productSubtotal", "totalAmount",
          "totalIsComplete", "whatsappStatus"
        ])
        && request.resource.data.orderNumber is string
        && request.resource.data.orderNumber.matches("^DW-[0-9]{8}$")
        && request.resource.data.status == "new"
        && request.resource.data.source == "website"
        && request.resource.data.createdAt == request.time
        && request.resource.data.customerName is string
        && request.resource.data.customerName.size() > 0
        && request.resource.data.customerName.size() <= 80
        && request.resource.data.customerPhone is string
        && request.resource.data.customerPhone.matches("^[0-9]{8,15}$")
        && request.resource.data.customerAddress is string
        && request.resource.data.customerAddress.size() > 0
        && request.resource.data.customerAddress.size() <= 300
        && request.resource.data.notes is string
        && request.resource.data.notes.size() <= 300
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 50
        && request.resource.data.totalText is string
        && request.resource.data.totalText.size() <= 1000
        && (
          !request.resource.data.keys().hasAny([
            "productSubtotal", "totalAmount", "totalIsComplete", "whatsappStatus"
          ])
          || (
            request.resource.data.keys().hasAll([
              "productSubtotal", "totalAmount", "totalIsComplete", "whatsappStatus"
            ])
            && request.resource.data.orderNumber == orderId
            && request.resource.data.whatsappStatus == "awaiting_customer_send"
            && request.resource.data.productSubtotal is number
            && request.resource.data.productSubtotal >= 0
            && request.resource.data.totalIsComplete is bool
            && (request.resource.data.totalAmount == null
              || (request.resource.data.totalAmount is number && request.resource.data.totalAmount >= 0))
          )
        );
      allow read, update, delete: if isStoreAdmin();
    }
  }
}
